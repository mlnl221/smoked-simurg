### 3. Python code

```python
import requests

API_KEY = "YOUR_OPENALEX_API_KEY"


def find_issn(magazine_name):
    url = "https://api.openalex.org/sources"

    params = {
        "search": magazine_name,
        "api_key": API_KEY,
        "per_page": 10
    }

    response = requests.get(url, params=params)
    response.raise_for_status()

    data = response.json()

    results = data.get("results", [])

    if not results:
        print("No results found.")
        return []

    matches = []

    for source in results:
        matches.append({
            "name": source.get("display_name"),
            "issn": source.get("issn", []),
            "issn_l": source.get("issn_l"),
            "type": source.get("type"),
        })

    return matches


# Example
magazine = "National Geographic"

results = find_issn(magazine)

for result in results:
    print(f"Name: {result['name']}")
    print(f"ISSN: {result['issn']}")
    print(f"ISSN-L: {result['issn_l']}")
    print(f"Type: {result['type']}")
    print("-" * 40)
```

The `/sources` endpoint specifically returns fields including `display_name`, `issn`, and `issn_l`, and supports searching by title. ([OpenAlex][3])

### 4. If you want just the best ISSN

You could simplify it to:

```python
import requests

API_KEY = "YOUR_OPENALEX_API_KEY"


def get_issn(magazine_name):
    response = requests.get(
        "https://api.openalex.org/sources",
        params={
            "search": magazine_name,
            "api_key": API_KEY,
            "per_page": 5
        }
    )

    response.raise_for_status()

    results = response.json()["results"]

    if not results:
        return None

    best_match = results[0]

    return {
        "name": best_match["display_name"],
        "issn": best_match.get("issn", []),
        "issn_l": best_match.get("issn_l")
    }


print(get_issn("National Geographic"))
```
